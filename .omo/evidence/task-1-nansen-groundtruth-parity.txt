  Prints exactly the 13 expected counts. On unreadable input: clear stderr
  message + exit 1 (no silent 0, no raw traceback)."""
  import json, collections, sys

  if len(sys.argv) != 2:
      sys.stderr.write("usage: count_groundtruth.py <path-to-tx-sample.jsonl>\n")
      sys.exit(2)
  P = sys.argv[1]

  try:
      rows = [json.loads(l) for l in open(P) if l.strip()]
  except FileNotFoundError:
      sys.stderr.write(f"ERROR: input file not found: {P}\n")
      sys.exit(1)
  except (OSError, json.JSONDecodeError) as e:
      sys.stderr.write(f"ERROR: cannot read {P}: {e}\n")
      sys.exit(1)

  c = collections.Counter(r["txType"] for r in rows)
  g = collections.defaultdict(list)
  sigs = collections.defaultdict(list)
  for r in rows:
      g[(r["transactionHash"], r["wallet"], r["ca"])].append(r)
      sigs[r["transactionHash"]].append(r)

  buy_only = sell_only = tr_only = buysell = bs_tr = usd_null = 0
  for v in g.values():
      kinds = set(r["txType"] for r in v)
      if kinds == {"buy"}:
          buy_only += 1
      elif kinds == {"sell"}:
          sell_only += 1
      elif kinds == {"transfer"}:
          tr_only += 1
      if "buy" in kinds and "sell" in kinds:
          buysell += 1
      if "buy" in kinds and "sell" in kinds and "transfer" in kinds:
          bs_tr += 1
      if any(r.get("usdValueAtTxTime") is None for r in v):
          usd_null += 1

  print("TOTAL ROWS:", len(rows))
  print(f"1. rows by txType: buy={c['buy']} sell={c['sell']} transfer={c['transfer']}")
  print("2. unique (transactionHash,wallet,ca) groups:", len(g))
  print("3. unique signatures:", len(sigs))
  print("4. buy-only groups:", buy_only)
  print("5. sell-only groups:", sell_only)
  print("6. transfer-only groups:", tr_only)
  print("7. buy+sell groups:", buysell)
  print("8. groups with usdValueAtTxTime null:", usd_null)
  print("9. field count sorted(rows[0].keys()):", len(sorted(rows[0].keys())))
  print("   fields:", sorted(rows[0].keys()))
  print("10. transfer rows with directionalAmountOfTokens != 0:",
        sum(1 for r in rows if r["txType"] == "transfer" and r.get("directionalAmountOfTokens") != 0))
  print("11. groups mixing buy/sell with transfer:", bs_tr)
  print("12. sigs with >1 wallet:", sum(1 for v in sigs.values() if len(set(r["wallet"] for r in v)) > 1))
  print("13. transfer-only sigs:", sum(1 for v in sigs.values() if all(r["txType"] == "transfer" for r in v)))
