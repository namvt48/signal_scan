## New SQL

allTokenStates (db.ts):
    SELECT * FROM token_state
    -- Map key: `${chain}:${ca}` (raw row values); lookup `${c.chain}:${c.address}` —
    -- exact parity with getTokenState's `WHERE ca = canonicalCa(?) AND chain = ?`.

sumHoldingAmountByCa (signals.ts):
    SELECT s.ca AS ca, COALESCE(SUM(s.token_amount), 0) AS total FROM wallet_token_state s
      WHERE EXISTS (SELECT 1 FROM wallet_trades t
                     WHERE t.wallet_id = s.wallet_id AND t.ca = s.ca
                       AND t.side = 'buy' AND t.source = 'watch')
      GROUP BY s.ca

trackedWalletStatsByCa (signals.ts):
    WITH members AS (
      SELECT DISTINCT b.ca AS ca, b.wallet_id AS wallet_id
        FROM wallet_trades b
       WHERE b.side = 'buy' AND b.source = 'watch' AND b.ts >= @memberSince)
    SELECT m.ca AS ca, w.name AS name, w.clan AS clan, w.tags AS tags,
           COALESCE(SUM(CASE WHEN t.side = 'buy' THEN t.amount_usd ELSE -t.amount_usd END), 0) AS inflow,
           SUM(CASE WHEN t.side = 'buy' THEN 1 ELSE 0 END) AS buys,
           SUM(CASE WHEN t.side = 'sell' THEN 1 ELSE 0 END) AS sells,
           COALESCE(MAX(t.ts), 0) AS lastTs,
           MAX(s.balance_usd) AS balUsd
      FROM members m
      JOIN wallets w ON w.id = m.wallet_id
      LEFT JOIN wallet_trades t
        ON t.wallet_id = m.wallet_id AND t.ca = m.ca AND t.source = 'watch' AND t.ts >= @statSince
      LEFT JOIN wallet_token_state s ON s.wallet_id = m.wallet_id AND s.ca = m.ca
     GROUP BY m.ca, m.wallet_id
     ORDER BY m.ca, lastTs DESC, w.name
